set(BVH2_TINYBVH_INCLUDE_DIR "" CACHE PATH "External TinyBVH checkout")
option(BVH2_BENCH_NATIVE "Tune comparison benchmark for this CPU" OFF)
if(NOT EXISTS "${BVH2_TINYBVH_INCLUDE_DIR}/tiny_bvh.h")
    message(FATAL_ERROR "Set BVH2_TINYBVH_INCLUDE_DIR to an external TinyBVH checkout")
endif()

find_package(Threads REQUIRED)
add_executable(compare_tiny compare_tiny.cpp)
target_link_libraries(compare_tiny PRIVATE BVH2 Threads::Threads)
target_include_directories(compare_tiny SYSTEM PRIVATE "${BVH2_TINYBVH_INCLUDE_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(compare_tiny PRIVATE -Wall -Wextra -ffp-contract=off)
    if(BVH2_BENCH_NATIVE)
        target_compile_options(compare_tiny PRIVATE -march=native)
    elseif(BVH2_ENABLE_AVX2)
        include(CheckCXXCompilerFlag)
        check_cxx_compiler_flag("-mavx2" BVH2_BENCH_HAS_AVX2)
        if(BVH2_BENCH_HAS_AVX2)
            target_compile_options(compare_tiny PRIVATE -mavx2)
        endif()
    endif()
elseif(MSVC)
    target_compile_options(compare_tiny PRIVATE /W4)
    if(BVH2_ENABLE_AVX2)
        target_compile_options(compare_tiny PRIVATE /arch:AVX2)
    endif()
endif()
